cmake_minimum_required(VERSION 3.16)
project(anchor_relay LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
option(UWB_SANITIZE "ASan/UBSan" OFF)
if(UWB_SANITIZE)
 add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
 add_link_options(-fsanitize=address,undefined)
endif()
set(UWB_STANDALONE ON CACHE BOOL "" FORCE)
add_subdirectory(../../packages/liuo_uwb_protocol protocol)
add_library(anchor_relay src/relay.cpp ../../packages/nlink_parser/src/ros2/framer.cpp)
target_include_directories(anchor_relay PUBLIC include ../../packages/nlink_parser/include)
target_link_libraries(anchor_relay PUBLIC liuo_uwb_protocol)
add_executable(relay_simulation tests/simulation.cpp)
target_link_libraries(relay_simulation anchor_relay)
target_compile_options(relay_simulation PRIVATE -UNDEBUG)
enable_testing()
add_test(NAME relay_simulation COMMAND relay_simulation)
