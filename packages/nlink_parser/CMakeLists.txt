cmake_minimum_required(VERSION 3.16)
project(nlink_parser LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(catkin REQUIRED COMPONENTS roscpp std_msgs diagnostic_msgs message_generation)
file(GLOB msg_files RELATIVE ${CMAKE_CURRENT_SOURCE_DIR}/msg msg/*.msg)
add_message_files(FILES ${msg_files})
generate_messages(DEPENDENCIES std_msgs)
catkin_package(CATKIN_DEPENDS roscpp std_msgs diagnostic_msgs message_runtime)
include_directories(${catkin_INCLUDE_DIRS})
add_library(uwb_serial STATIC extern/serial/src/serial.cc extern/serial/src/impl/unix.cc extern/serial/src/impl/list_ports/list_ports_linux.cc)
target_include_directories(uwb_serial PUBLIC extern/serial/include)
target_link_libraries(uwb_serial pthread rt)
file(GLOB unpack_sources src/utils/nlink_unpack/nlink_linktrack_*.c)
list(FILTER unpack_sources EXCLUDE REGEX "aoa")
add_library(nlink_core STATIC src/ros2/framer.cpp src/utils/nlink_unpack/nlink_utils.c ${unpack_sources})
target_include_directories(nlink_core PUBLIC include src/utils/nlink_unpack)
add_executable(linktrack src/ros1/driver.cpp src/ros1/publish.cpp)
target_link_libraries(linktrack nlink_core uwb_serial)
target_link_libraries(linktrack ${catkin_LIBRARIES})
add_dependencies(linktrack ${${PROJECT_NAME}_EXPORTED_TARGETS} ${catkin_EXPORTED_TARGETS})
install(TARGETS linktrack RUNTIME DESTINATION ${CATKIN_PACKAGE_BIN_DESTINATION})
install(DIRECTORY launch DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION})
