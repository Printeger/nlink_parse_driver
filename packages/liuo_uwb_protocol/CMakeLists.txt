cmake_minimum_required(VERSION 3.16)
project(liuo_uwb_protocol LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
option(UWB_STANDALONE "Build without ROS" OFF)
option(UWB_SANITIZE "Enable ASan and UBSan" OFF)
if(UWB_SANITIZE)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
if(NOT UWB_STANDALONE)
  find_package(catkin REQUIRED)
  catkin_package(INCLUDE_DIRS include LIBRARIES ${PROJECT_NAME})

endif()

add_library(${PROJECT_NAME} src/codec.cpp)
target_include_directories(${PROJECT_NAME} PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_compile_options(${PROJECT_NAME} PRIVATE -Wall -Wextra -Wpedantic)
include(CTest)
if(BUILD_TESTING)
  add_executable(protocol_golden tests/golden.cpp)
  target_link_libraries(protocol_golden ${PROJECT_NAME})
  target_compile_options(protocol_golden PRIVATE -UNDEBUG)
  add_test(NAME protocol_golden COMMAND protocol_golden)
  add_executable(codec_cli tests/codec_cli.cpp)
  target_link_libraries(codec_cli ${PROJECT_NAME})
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_test(NAME protocol_crosscheck COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/crosscheck.py $<TARGET_FILE:codec_cli>)
  set_tests_properties(protocol_crosscheck PROPERTIES ENVIRONMENT "PYTHONDONTWRITEBYTECODE=1")
endif()
install(TARGETS ${PROJECT_NAME} EXPORT protocolTargets ARCHIVE DESTINATION lib LIBRARY DESTINATION lib)
install(DIRECTORY include/ DESTINATION include)
install(EXPORT protocolTargets DESTINATION share/${PROJECT_NAME}/cmake)
