cmake_minimum_required(VERSION 3.16)
project(liuo_uwb_bridge LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(nlink_parser REQUIRED)
find_package(liuo_uwb_msgs REQUIRED)
find_package(liuo_uwb_protocol REQUIRED)
find_package(diagnostic_msgs REQUIRED)
find_package(std_srvs REQUIRED)
find_package(yaml-cpp REQUIRED)
add_executable(range_bridge src/bridge.cpp)
target_include_directories(range_bridge PRIVATE include)
target_link_libraries(range_bridge liuo_uwb_protocol::liuo_uwb_protocol yaml-cpp)
ament_target_dependencies(range_bridge rclcpp nlink_parser liuo_uwb_msgs diagnostic_msgs std_srvs)
install(TARGETS range_bridge DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY config launch DESTINATION share/${PROJECT_NAME})
ament_package()
if(BUILD_TESTING)
  enable_testing()
  add_executable(state_test test/state_test.cpp)
  target_include_directories(state_test PRIVATE include)
  target_compile_options(state_test PRIVATE -UNDEBUG)
  add_test(NAME state_test COMMAND state_test)
endif()
