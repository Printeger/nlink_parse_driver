cmake_minimum_required(VERSION 3.16)
project(liuo_uwb_bridge LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(catkin REQUIRED COMPONENTS roscpp nlink_parser liuo_uwb_msgs liuo_uwb_protocol diagnostic_msgs std_srvs)
find_package(yaml-cpp REQUIRED)
catkin_package(CATKIN_DEPENDS roscpp nlink_parser liuo_uwb_msgs liuo_uwb_protocol diagnostic_msgs std_srvs)
add_executable(range_bridge src/bridge.cpp)
target_include_directories(range_bridge PRIVATE include ${catkin_INCLUDE_DIRS})
target_link_libraries(range_bridge ${catkin_LIBRARIES} yaml-cpp)
add_dependencies(range_bridge ${catkin_EXPORTED_TARGETS})
install(TARGETS range_bridge RUNTIME DESTINATION ${CATKIN_PACKAGE_BIN_DESTINATION})
catkin_install_python(PROGRAMS scripts/observation_reader.py DESTINATION ${CATKIN_PACKAGE_BIN_DESTINATION})
install(DIRECTORY config launch DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION})
if(CATKIN_ENABLE_TESTING)
  add_executable(state_test test/state_test.cpp)
  target_include_directories(state_test PRIVATE include)
  target_compile_options(state_test PRIVATE -UNDEBUG)
  add_test(NAME state_test COMMAND state_test)
endif()
